# 11-a - Quantas doenças estão presentes na ontologia?
# ==============================================================================

PREFIX : <http://www.example.org/disease-ontology#>
SELECT (COUNT(DISTINCT ?disease) AS ?totalDoencas)
WHERE {
    ?disease a :Disease .
}


# ==============================================================================
# 11-b - Que doenças estão associadas ao sintoma "yellowish_skin"?
# ==============================================================================

PREFIX : <http://www.example.org/disease-ontology#>
SELECT ?disease
WHERE {
    ?disease a :Disease ;
             :hasSymptom :yellowish_skin ;
}

# ==============================================================================
# 12 - CONSTRUCT: diagnosticar a doença de cada doente
#      Lógica: associar o doente à doença com mais sintomas em comum.
#              Gera triplos da forma :patientX :hasDisease :diseaseY
# ==============================================================================

PREFIX : <http://www.example.org/disease-ontology#>
CONSTRUCT {
    ?patient :hasDisease ?disease .
}
WHERE {
    ?patient a :Patient .
    ?disease a :Disease .
    ?patient :exhibitsSymptom ?symptom .
    ?disease :hasSymptom ?symptom .
}

PREFIX : <http://www.example.org/disease-ontology#>
INSERT {
    ?patient :hasDisease ?disease .
}
WHERE {
    ?patient a :Patient .
    ?disease a :Disease .
    ?patient :exhibitsSymptom ?symptom .
    ?disease :hasSymptom ?symptom .
}


# ==============================================================================
# 13 - Distribuição dos doentes pelas doenças (doença, nº de doentes)
#      Nota: requer que os triplos :hasDisease gerados na Q5 estejam na ontologia
# ==============================================================================

PREFIX : <http://www.example.org/disease-ontology#>
SELECT ?nomeDoenca (COUNT(DISTINCT ?patient) AS ?numDoentes)
WHERE {
    ?patient a :Patient ;
             :hasDisease ?disease .
    ?disease :name ?nomeDoenca .
}
GROUP BY ?nomeDoenca
ORDER BY DESC(?numDoentes)

# ==============================================================================
# 16 - Lista das doenças e respetivos sintomas (doença, lista de sintomas)
# ==============================================================================

PREFIX : <http://www.example.org/disease-ontology#>
SELECT ?nomeDoenca (GROUP_CONCAT(?nomeSintoma ; separator=", ") AS ?sintomas)
WHERE {
    ?disease a :Disease ;
             :name ?nomeDoenca ;
             :hasSymptom ?symptom .
    ?symptom :name ?nomeSintoma .
}
GROUP BY ?nomeDoenca
ORDER BY ?nomeDoenca
